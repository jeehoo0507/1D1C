```python
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim

x = torch.FloatTensor([[1],[2],[3]])
y = torch.FloatTensor([[2],[4],[6]])

model = nn.Linear()
optimizer = optim.SGD(model.parameters(), lr=0.01)

lambda_reg= 0.1  

nb_epoch = 1000
for epoch in range(1, nb_epoch + 1):
    hypothesis = model(x)
    mse = F.mse_loss(hypothesis, y)

    W = model.weight

    # L2
    l2 = lambda_reg * torch.sum(W ** 2)
    # L1
    l1 = lambda_reg * torch.sum(torch.abs(W))

    cost = mse + l2 # l1 / l2
   

    optimizer.zero_grad()
    cost.backward()
    optimizer.step()

    if epoch % 100 == 0:
        params = list(model.parameters())
        W_val, b_val = params[0], params[1]
        print(f"epoch {epoch:4d}/{nb_epoch} W: {W_val.item():.3f} b: {b_val.item():.3f} cost: {cost.item():.6f}")
```
