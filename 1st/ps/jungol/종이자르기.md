```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main(){
    int n, m;

    cin >> n >>m ;
    
    int T;
    cin>>T;

    vector<int> arr_zero; 
    vector<int> arr_one;  


    while(T--){
        int a, b;
        cin >> a>>b;
        if(a==0){
            arr_zero.push_back(b);
        }
        else arr_one.push_back(b);
    }

    arr_zero.push_back(m);
    arr_one.push_back(n);

    sort(arr_one.begin(), arr_one.end());
    sort(arr_zero.begin(), arr_zero.end());

    int zero_max = arr_zero[0];
    int one_max = arr_one[0];

    for(int i = 0;i<arr_zero.size();i++){
        if (zero_max < (arr_zero[i+1] - arr_zero[i])){
            zero_max = arr_zero[i+1] - arr_zero[i];
        }
    }
    for(int i = 0;i<arr_one.size();i++){
        if (one_max < (arr_one[i+1] - arr_one[i])){
            one_max = arr_one[i+1] - arr_one[i];
        }
    }

    cout << zero_max * one_max;


    return 0;
}
```
